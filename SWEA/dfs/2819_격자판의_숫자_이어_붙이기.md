# 2819. 격자판의 숫자 이어 붙이기

## Code (Python)
```python3
dxs = [1, 0, -1, 0]
dys = [0, 1, 0, -1]

def dfs(y, x, cur, path, grid, seen):
    if cur == 7:
        seen.add(path)
        return

    for dx, dy in zip(dxs, dys):
        ax, ay = x + dx, y + dy
        if 0 <= ax < 4 and 0 <= ay < 4:
            dfs(ay, ax, cur + 1, path + str(grid[ay][ax]), grid, seen)

T = int(input())
for test_case in range(1, T + 1):
    grid = [list(map(int, input().split())) for _ in range(4)]
    seen = set()
    for i in range(4):
        for j in range(4):
            dfs(i, j, 1, str(grid[i][j]), grid, seen)
    print(f"#{test_case} {len(seen)}")      
```
