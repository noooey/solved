# 1868. 파핑파핑 지뢰찾기

## Code (Python)
```python3
from collections import deque
axs = [1, 1, 1, 0, -1, -1, -1, 0]
ays = [-1, 0, 1, 1, 1, 0, -1, -1]

T = int(input())
# 여러개의 테스트 케이스가 주어지므로, 각각을 처리합니다.
for test_case in range(1, T + 1):
    N = int(input())
    grid = [list(input().strip()) for _ in range(N)]
    cnts = [[0] * N for _ in range(N)]

    for i in range(N):
        for j in range(N):
            cnt = 0
            for ay, ax in zip(ays, axs):
                ny, nx = i + ay, j + ax
                if 0 <= ny < N and 0 <= nx < N:
                    if grid[ny][nx] == '*':
                        cnt += 1
            cnts[i][j] = cnt

    visited = [[False] * N for _ in range(N)]
    clicks = 0

    def bfs(sy, sx):
        q = deque([(sy, sx)])
        visited[sy][sx] = True
        while q:
            y, x = q.popleft()
            if cnts[y][x] == 0:
                for ay, ax in zip(ays, axs):
                    ny, nx = y + ay, x + ax
                    if 0 <= ny < N and 0 <= nx < N:
                        if grid[ny][nx] == '.' and not visited[ny][nx]:
                            visited[ny][nx] = True
                            if cnts[ny][nx] == 0:
                                q.append((ny, nx))

    for i in range(N):
        for j in range(N):
            if grid[i][j] == '.' and cnts[i][j] == 0 and not visited[i][j] :
                clicks += 1
                bfs(i, j)

    for i in range(N):
        for j in range(N):
            if grid[i][j] == '.' and not visited[i][j] :
                clicks += 1

    print(f"#{test_case} {clicks}")
```

## Complexity
- Time: O(n**2)
