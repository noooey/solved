# 1226. [S/W 문제해결 기본] 7일차 - 미로1

## Code (Python)
```python3
from collections import deque
T = 10
axs = [1, 0, -1, 0]
ays = [0, 1, 0, -1]
# 여러개의 테스트 케이스가 주어지므로, 각각을 처리합니다.
for _ in range(1, T + 1):
    test_case = int(input())
    n = 16
    grid = [list(map(int, input().strip())) for _ in range(n)]
    start = (-1, -1)
    for i in range(n):
        for j in range(n):
            if grid[i][j] == 2:
                start = (i, j)
                grid[i][j] = 1
                break
        if start != (-1, -1):
            break
    queue = deque([start])
    result = 0
    while queue:
        y, x = queue.popleft()
        for ay, ax in zip(ays, axs):
            ny, nx = y + ay, x + ax
            if not (0 <= ny < n and 0 <= nx < n):
                continue
            if grid[ny][nx] == 3:
                result = 1
                queue.clear()
                break
            if grid[ny][nx] == 0:
                grid[ny][nx] = 1
                queue.append((ny, nx))
    print(f"#{test_case} {result}")
```

## Complexity
- Time: O(n**2)
