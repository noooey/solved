# 1206. [S/W 문제해결 기본] 1일차 - View

## Code (Python)
```python3
T = 10
# 여러개의 테스트 케이스가 주어지므로, 각각을 처리합니다.
axs = [-2, -1, 1, 2]

for test_case in range(1, T + 1):
    n = int(input())
    buildings = list(map(int, input().split()))

    grid = [[0] * n for _ in range(255)]

    for i in range(n):
        height = buildings[i]
        for j in range(height):
            grid[j][i] = 1
        cnt = 0
    for i in range(255):
        for j in range(n):
            if grid[i][j] != 1:
                continue

            is_exist = False
            for ax in axs:
                nx = j + ax
                if 0 <= nx < n:
                    if grid[i][nx] == 1:
                        is_exist = True
                        break
            if not is_exist:
                cnt += 1

    print(f"#{test_case} {cnt}")
```

## Complexity
- Time: O(nm)
