## Code (Python)
```python3
import heapq
#sys.stdin = open("input.txt", "r")

T = int(input())
INF = 10**18
dxs = [1, 0, -1, 0]
dys = [0, -1, 0, 1]
# 여러개의 테스트 케이스가 주어지므로, 각각을 처리합니다.
def find_minimum(n, grid):
    cost = [[INF] * n for _ in range(n)]
    cost[0][0] = 0
    pq = [(0, 0, 0)]   # cost, x, y

    while pq:
        c, x, y = heapq.heappop(pq)
        if c > cost[y][x]:
            continue
        if x == n - 1 and y == n - 1:
            return c

        for dx, dy in zip(dxs, dys):
            ax, ay = x + dx, y + dy
            if 0 <= ax < n and 0 <= ay < n:
                ncost = c + grid[ay][ax]
                if ncost < cost[ay][ax]:
                    cost[ay][ax] = ncost
                    heapq.heappush(pq, (ncost, ax, ay))

for test_case in range(1, T + 1):
    # ///////////////////////////////////////////////////////////////////////////////////
    n = int(input())
    grid = []
    for _ in range(n):
        grid.append(list(map(int, input())))
    print(f"#{test_case} {find_minimum(n, grid)}")
    # ///////////////////////////////////////////////////////////////////////////////////
```

## Complexity
- Time: O(n**2logn)
